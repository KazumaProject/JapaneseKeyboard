cmake_minimum_required(VERSION 3.22.1)
project(zenz_model_compat_test LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(LLAMA_BUILD_COMMON OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_MTMD OFF CACHE BOOL "" FORCE)
set(GGML_METAL OFF CACHE BOOL "" FORCE)
set(GGML_BLAS OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
add_subdirectory(../../main/cpp/llama.cpp llama)
add_executable(zenz_model_compat_test zenz_model_compat_test.cpp)
target_include_directories(zenz_model_compat_test PRIVATE ../../main/cpp)
target_link_libraries(zenz_model_compat_test PRIVATE llama)
enable_testing()
add_test(NAME zenz_model_compat COMMAND zenz_model_compat_test)
